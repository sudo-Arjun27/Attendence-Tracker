import 'package:flutter/material.dart';
import 'package:shared_preferences/shared_preferences.dart';
import 'package:table_calendar/table_calendar.dart';
import 'dart:convert';

import 'main.dart';

class AttendanceStudent extends StatefulWidget {
  final User user;
  const AttendanceStudent({super.key, required this.user});

  @override
  _AttendanceStudentState createState() => _AttendanceStudentState();
}

class _AttendanceStudentState extends State<AttendanceStudent> {
  DateTime selectedDate = DateTime.now();
  String? selectedSubject;
  String? statusMessage;
  String studentClass = '';
  Map<DateTime, bool> userAttendanceMap = {};

  final List<String> subjects = [
    'Maths',
    'Digital Electronics',
    'Physics',
    'Programming Logic Skills',
    'Python',
  ];

  @override
  void initState() {
    super.initState();
    selectedSubject = subjects.first;
    studentClass = _getClassByStudent(widget.user.username);
    _loadAttendanceForSubject();
  }

  String _getClassByStudent(String username) {
    if (['student1', 'student2'].contains(username)) return 'Class A';
    return 'Class B';
  }

  void _loadAttendanceForSubject() async {
    if (selectedSubject == null) return;

    final prefs = await SharedPreferences.getInstance();
    Map<DateTime, bool> tempMap = {};

    for (int i = 0; i < 365; i++) {
      DateTime date = DateTime.now().subtract(Duration(days: i));
      final key = '${selectedSubject}_${studentClass}_${_formatDate(date)}';
      final data = prefs.getString('attendance_$key');
      if (data != null) {
        final parsed = json.decode(data) as Map<String, dynamic>;
        if (parsed.containsKey(widget.user.username)) {
          tempMap[DateTime(date.year, date.month, date.day)] =
              parsed[widget.user.username];
        }
      }
    }

    setState(() {
      userAttendanceMap = tempMap;
      _loadAttendanceStatusForSelectedDate();
    });
  }

  void _loadAttendanceStatusForSelectedDate() async {
    if (selectedSubject == null) return;

    final prefs = await SharedPreferences.getInstance();
    final key = '${selectedSubject}_${studentClass}_${_formattedDate()}';
    final data = prefs.getString('attendance_$key');
    if (data != null) {
      final parsed = json.decode(data) as Map<String, dynamic>;
      final status = parsed[widget.user.username];
      setState(() {
        if (status == true) {
          statusMessage = 'You were PRESENT on this day for $selectedSubject.';
        } else if (status == false) {
          statusMessage = 'You were ABSENT on this day for $selectedSubject.';
        } else {
          statusMessage = 'No attendance record found.';
        }
      });
    } else {
      setState(() {
        statusMessage = 'No attendance recorded yet.';
      });
    }
  }

  String _formattedDate() => selectedDate.toIso8601String().split("T")[0];
  String _formatDate(DateTime dt) => dt.toIso8601String().split("T")[0];

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: LayoutBuilder(
        builder: (context, constraints) {
          bool isWide = constraints.maxWidth > 600;

          return SingleChildScrollView(
            padding: EdgeInsets.all(16),
            child:
                isWide
                    ? Row(
                      crossAxisAlignment: CrossAxisAlignment.start,
                      children: [
                        Expanded(child: _buildSubjectDropdown()),
                        SizedBox(width: 20),
                        Expanded(child: _buildCalendar()),
                        SizedBox(width: 20),
                        Expanded(child: _buildStatusMessage()),
                      ],
                    )
                    : Column(
                      crossAxisAlignment: CrossAxisAlignment.stretch,
                      children: [
                        _buildSubjectDropdown(),
                        SizedBox(height: 16),
                        _buildCalendar(),
                        SizedBox(height: 16),
                        _buildStatusMessage(),
                      ],
                    ),
          );
        },
      ),
    );
  }

  Widget _buildSubjectDropdown() {
    return DropdownButtonFormField<String>(
      value: selectedSubject,
      decoration: InputDecoration(
        labelText: "Select Subject",
        border: OutlineInputBorder(),
      ),
      items:
          subjects
              .map(
                (subject) =>
                    DropdownMenuItem(value: subject, child: Text(subject)),
              )
              .toList(),
      onChanged: (newSubject) {
        setState(() {
          selectedSubject = newSubject!;
          userAttendanceMap.clear();
        });
        _loadAttendanceForSubject();
      },
    );
  }

  Widget _buildCalendar() {
    return TableCalendar(
      focusedDay: selectedDate,
      firstDay: DateTime(2020),
      lastDay: DateTime.now(),
      calendarFormat: CalendarFormat.month,
      selectedDayPredicate: (day) => isSameDay(selectedDate, day),
      onDaySelected: (selected, _) {
        setState(() {
          selectedDate = selected;
        });
        _loadAttendanceStatusForSelectedDate();
      },
      calendarBuilders: CalendarBuilders(
        defaultBuilder: (context, day, _) {
          final attendance =
              userAttendanceMap[DateTime(day.year, day.month, day.day)];
          if (attendance != null) {
            return Container(
              margin: EdgeInsets.all(6),
              decoration: BoxDecoration(
                color:
                    attendance
                        ? Colors.green.withOpacity(0.3)
                        : Colors.red.withOpacity(0.3),
                shape: BoxShape.circle,
              ),
              alignment: Alignment.center,
              child: Text(
                '${day.day}',
                style: TextStyle(
                  color: Colors.black,
                  fontWeight: FontWeight.w600,
                ),
              ),
            );
          }
          return null;
        },
        todayBuilder: (context, day, _) {
          return Container(
            margin: EdgeInsets.all(6),
            decoration: BoxDecoration(
              color: Colors.blue.withOpacity(0.5),
              shape: BoxShape.circle,
            ),
            alignment: Alignment.center,
            child: Text(
              '${day.day}',
              style: TextStyle(
                color: Colors.white,
                fontWeight: FontWeight.bold,
              ),
            ),
          );
        },
      ),
    );
  }

  Widget _buildStatusMessage() {
    return Padding(
      padding: const EdgeInsets.symmetric(vertical: 16),
      child: Text(
        statusMessage ?? '',
        style: TextStyle(
          fontSize: 18,
          color:
              statusMessage?.contains('PRESENT') == true
                  ? Colors.green
                  : statusMessage?.contains('ABSENT') == true
                  ? Colors.red
                  : Colors.black,
        ),
        textAlign: TextAlign.center,
      ),
    );
  }
}
